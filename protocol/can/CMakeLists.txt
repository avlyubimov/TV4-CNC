cmake_minimum_required(VERSION 3.20)
project(tv4_update_transport C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(tv4_update_transport STATIC update_session.c update_outcome.c update_negotiation.c)
target_include_directories(tv4_update_transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(tv4_update_transport PRIVATE
  -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
add_library(tv4_update_wire STATIC update_wire.c)
target_include_directories(tv4_update_wire PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tv4_update_wire PUBLIC tv4_update_transport)
target_compile_options(tv4_update_wire PRIVATE
  -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
if(CMAKE_CROSSCOMPILING)
  target_compile_options(tv4_update_transport PRIVATE -fstack-usage -mgeneral-regs-only)
  target_compile_options(tv4_update_wire PRIVATE -fstack-usage -mgeneral-regs-only)
endif()
if(NOT CMAKE_CROSSCOMPILING)
  enable_testing()
  find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
  add_executable(update_session_tests tests/session_tests.c)
  target_link_libraries(update_session_tests PRIVATE tv4_update_transport)
  add_executable(update_loopback_tests tests/loopback_tests.c)
  target_link_libraries(update_loopback_tests PRIVATE tv4_update_transport OpenSSL::Crypto)
  add_executable(update_outcome_tests tests/outcome_tests.c)
  target_link_libraries(update_outcome_tests PRIVATE tv4_update_transport)
  add_executable(update_negotiation_tests tests/negotiation_tests.c)
  target_link_libraries(update_negotiation_tests PRIVATE tv4_update_transport)
  add_executable(update_wire_tests tests/wire_tests.c)
  target_link_libraries(update_wire_tests PRIVATE tv4_update_wire)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME update_wire_reference COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/wire_reference.py" --driver $<TARGET_FILE:update_wire_tests>)
  set_tests_properties(update_wire_reference PROPERTIES TIMEOUT 15)
  foreach(test IN ITEMS update_session update_loopback update_outcome update_negotiation update_wire)
    target_compile_options(${test}_tests PRIVATE
      -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -UNDEBUG)
    add_test(NAME ${test} COMMAND ${test}_tests)
    set_tests_properties(${test} PROPERTIES TIMEOUT 20)
  endforeach()
endif()
