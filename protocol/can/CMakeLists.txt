cmake_minimum_required(VERSION 3.20)
project(tv4_update_transport C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(tv4_update_transport STATIC update_session.c update_outcome.c update_negotiation.c)
target_include_directories(tv4_update_transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(tv4_update_transport PRIVATE
  -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
add_library(tv4_update_wire STATIC update_wire.c)
target_include_directories(tv4_update_wire PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tv4_update_wire PUBLIC tv4_update_transport)
target_compile_options(tv4_update_wire PRIVATE
  -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
add_library(tv4_update_fragments STATIC update_fragments.c)
add_library(tv4_update_metadata STATIC update_metadata.c)
add_library(tv4_metadata_setup STATIC update_metadata_setup.c)
target_include_directories(tv4_metadata_setup PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tv4_metadata_setup PUBLIC tv4_update_metadata)
target_compile_options(tv4_metadata_setup PRIVATE
  -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
target_include_directories(tv4_update_metadata PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tv4_update_metadata PUBLIC tv4_update_transport)
target_compile_options(tv4_update_metadata PRIVATE
  -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
target_include_directories(tv4_update_fragments PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tv4_update_fragments PUBLIC tv4_update_wire)
target_compile_options(tv4_update_fragments PRIVATE
  -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
add_library(tv4_update_channel STATIC update_channel.c)
target_include_directories(tv4_update_channel PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tv4_update_channel PUBLIC tv4_update_fragments tv4_metadata_setup)
target_compile_options(tv4_update_channel PRIVATE
  -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
if(CMAKE_CROSSCOMPILING)
  target_compile_options(tv4_update_channel PRIVATE -fstack-usage -mgeneral-regs-only)
  target_compile_options(tv4_update_transport PRIVATE -fstack-usage -mgeneral-regs-only)
  target_compile_options(tv4_update_wire PRIVATE -fstack-usage -mgeneral-regs-only)
  target_compile_options(tv4_update_fragments PRIVATE -fstack-usage -mgeneral-regs-only)
  target_compile_options(tv4_update_metadata PRIVATE -fstack-usage -mgeneral-regs-only)
  target_compile_options(tv4_metadata_setup PRIVATE -fstack-usage -mgeneral-regs-only)
endif()
if(NOT CMAKE_CROSSCOMPILING)
  enable_testing()
  find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
  add_executable(update_session_tests tests/session_tests.c)
  target_link_libraries(update_session_tests PRIVATE tv4_update_transport)
  add_executable(update_loopback_tests tests/loopback_tests.c)
  target_link_libraries(update_loopback_tests PRIVATE tv4_update_transport OpenSSL::Crypto)
  add_executable(update_outcome_tests tests/outcome_tests.c)
  target_link_libraries(update_outcome_tests PRIVATE tv4_update_transport)
  add_executable(update_negotiation_tests tests/negotiation_tests.c)
  target_link_libraries(update_negotiation_tests PRIVATE tv4_update_transport)
  add_executable(update_wire_tests tests/wire_tests.c)
  target_link_libraries(update_wire_tests PRIVATE tv4_update_wire)
  add_executable(update_fragment_tests tests/fragment_tests.c)
  target_link_libraries(update_fragment_tests PRIVATE tv4_update_fragments)
  add_library(tv4_fragment_test_link STATIC tests/fragment_test_link.c)
  target_include_directories(tv4_fragment_test_link PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/tests")
  target_link_libraries(tv4_fragment_test_link PUBLIC tv4_update_fragments tv4_metadata_setup tv4_update_channel)
  target_compile_options(tv4_fragment_test_link PRIVATE
    -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -UNDEBUG)
  add_executable(update_fragment_timing_tests tests/fragment_timing_tests.c)
  target_link_libraries(update_fragment_timing_tests PRIVATE tv4_fragment_test_link)
  add_executable(update_metadata_tests tests/metadata_tests.c)
  target_link_libraries(update_metadata_tests PRIVATE tv4_update_metadata)
  add_executable(update_channel_tests tests/channel_tests.c)
  target_link_libraries(update_channel_tests PRIVATE tv4_update_channel)
  add_executable(update_fragment_raw_tests tests/fragment_raw_tests.c)
  target_link_libraries(update_fragment_raw_tests PRIVATE tv4_update_fragments tv4_metadata_setup)
  add_executable(update_metadata_setup_tests tests/metadata_setup_tests.c)
  target_link_libraries(update_metadata_setup_tests PRIVATE tv4_metadata_setup)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME update_metadata_setup_reference COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/metadata_setup_reference.py" --driver $<TARGET_FILE:update_metadata_setup_tests>)
  set_tests_properties(update_metadata_setup_reference PROPERTIES TIMEOUT 15)
  add_test(NAME update_wire_reference COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/wire_reference.py" --driver $<TARGET_FILE:update_wire_tests>)
  set_tests_properties(update_wire_reference PROPERTIES TIMEOUT 15)
  add_test(NAME update_fragment_reference COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/fragment_reference.py" --driver $<TARGET_FILE:update_fragment_tests>)
  set_tests_properties(update_fragment_reference PROPERTIES TIMEOUT 15)
  add_test(NAME update_metadata_reference COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/metadata_reference.py" --driver $<TARGET_FILE:update_metadata_tests>)
  set_tests_properties(update_metadata_reference PROPERTIES TIMEOUT 15)
  foreach(test IN ITEMS update_session update_loopback update_outcome update_negotiation update_wire update_fragment update_fragment_timing update_metadata update_metadata_setup update_fragment_raw update_channel)
    target_compile_options(${test}_tests PRIVATE
      -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -UNDEBUG)
    add_test(NAME ${test} COMMAND ${test}_tests)
    set_tests_properties(${test} PROPERTIES TIMEOUT 20)
  endforeach()
endif()
