cmake_minimum_required(VERSION 3.22)
include($ENV{IDF_PATH}/tools/cmake/version.cmake)
set(TV4_IDF_VERSION "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}")
set(TV4_SUPPORTED_IDF "5.3.2" "6.1.1")
if(NOT TV4_IDF_VERSION IN_LIST TV4_SUPPORTED_IDF)
    message(FATAL_ERROR "TV4 validated SDKs: 5.3.2 / 6.1.1; found ${TV4_IDF_VERSION}")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_CURRENT_LIST_DIR}/locks/idf-${TV4_IDF_VERSION}.lock")
project(tv4_hmi)
