cmake_minimum_required(VERSION 3.20)
project(tv4_boot_admission C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(TV4_BOOT_BOARD_IMAGE "Build real bootloader (blocked)" OFF)
if(TV4_BOOT_BOARD_IMAGE)
  message(FATAL_ERROR "STM32 bootloader image blocked: board memory/erase/protection/recovery and physical quiescent contract are unreviewed")
endif()
include(FetchContent)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(tv4_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(mbed_lock "${tv4_root}/third_party/mbedtls/origin.json")
file(READ "${mbed_lock}" mbed_origin)
string(JSON mbed_url GET "${mbed_origin}" url)
string(JSON mbed_sha256 GET "${mbed_origin}" archive_sha256)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(TV4_BOOT_MBEDTLS_ARCHIVE "" CACHE FILEPATH "Optional official archive, still SHA-256 verified")
if(TV4_BOOT_MBEDTLS_ARCHIVE)
  set(mbed_url "${TV4_BOOT_MBEDTLS_ARCHIVE}")
endif()
FetchContent_Declare(tv4_boot_mbedtls URL "${mbed_url}"
  URL_HASH "SHA256=${mbed_sha256}"
  TLS_VERIFY TRUE
  SOURCE_SUBDIR tv4-no-upstream-build)
FetchContent_MakeAvailable(tv4_boot_mbedtls)
string(JSON mbed_count LENGTH "${mbed_origin}" crypto_sources)
math(EXPR mbed_last "${mbed_count} - 1")
set(mbed_files)
foreach(index RANGE ${mbed_last})
  string(JSON source GET "${mbed_origin}" crypto_sources ${index})
  list(APPEND mbed_files "${tv4_boot_mbedtls_SOURCE_DIR}/library/${source}.c")
endforeach()
add_custom_target(tv4_boot_source_check
  COMMAND ${Python3_EXECUTABLE} "${tv4_root}/tools/check_boot_crypto_source.py"
    --source "${tv4_boot_mbedtls_SOURCE_DIR}" --lock "${mbed_lock}"
  VERBATIM)
add_library(tv4_boot_crypto STATIC ${mbed_files})
add_dependencies(tv4_boot_crypto tv4_boot_source_check)
target_include_directories(tv4_boot_crypto PUBLIC "${tv4_boot_mbedtls_SOURCE_DIR}/include"
  "${CMAKE_CURRENT_SOURCE_DIR}/config")
target_compile_definitions(tv4_boot_crypto PUBLIC MBEDTLS_CONFIG_FILE="tv4_boot_mbedtls_config.h")
target_compile_options(tv4_boot_crypto PRIVATE -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
add_library(tv4_boot_admission STATIC core/admission.c core/metadata.c core/release_policy.c core/image_profile.c)
target_include_directories(tv4_boot_admission PUBLIC include PRIVATE core)
target_link_libraries(tv4_boot_admission PRIVATE tv4_boot_crypto)
target_compile_options(tv4_boot_admission PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
if(NOT TARGET tv4_update_transport)
  add_subdirectory("${tv4_root}/protocol/can" "${CMAKE_CURRENT_BINARY_DIR}/update-transport")
endif()
add_library(tv4_boot_handoff STATIC core/update_handoff.c core/update_completion.c core/release_handoff.c core/staging_source.c)
target_include_directories(tv4_boot_handoff PUBLIC include)
target_link_libraries(tv4_boot_handoff PUBLIC tv4_boot_admission tv4_update_transport)
target_compile_options(tv4_boot_handoff PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
if(CMAKE_CROSSCOMPILING)
  target_compile_options(tv4_boot_admission PRIVATE -fstack-usage -mgeneral-regs-only)
  target_compile_options(tv4_boot_crypto PRIVATE -fstack-usage -mgeneral-regs-only)
  target_compile_options(tv4_boot_handoff PRIVATE -fstack-usage -mgeneral-regs-only)
else()
  enable_testing()
  find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
  # Host filesystem/monotonic-clock frontend; never compiled into ARM firmware.
  add_executable(boot_motion_review tools/motion_review.c)
  target_link_libraries(boot_motion_review PRIVATE tv4_boot_admission)
  target_compile_definitions(boot_motion_review PRIVATE _POSIX_C_SOURCE=200809L)
  target_compile_options(boot_motion_review PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)

  add_executable(boot_admission_tests tests/admission_tests.c)
  target_link_libraries(boot_admission_tests PRIVATE tv4_boot_admission tv4_boot_crypto OpenSSL::Crypto)
  target_compile_options(boot_admission_tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -UNDEBUG)
  add_test(NAME boot_image_admission COMMAND boot_admission_tests)
  set_tests_properties(boot_image_admission PROPERTIES TIMEOUT 30)
  add_test(NAME boot_crypto_source COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/source_tests.py" --source "${tv4_boot_mbedtls_SOURCE_DIR}")
  set_tests_properties(boot_crypto_source PROPERTIES TIMEOUT 15)
  add_executable(boot_staging_tests tests/staging_tests.c)
  target_link_libraries(boot_staging_tests PRIVATE tv4_boot_handoff)
  target_compile_options(boot_staging_tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -UNDEBUG)
  add_test(NAME boot_staging_source COMMAND boot_staging_tests)
  set_tests_properties(boot_staging_source PROPERTIES TIMEOUT 30)
  add_executable(boot_handoff_tests tests/handoff_tests.c)
  target_link_libraries(boot_handoff_tests PRIVATE tv4_boot_handoff tv4_fragment_test_link tv4_update_metadata OpenSSL::Crypto)
  target_compile_options(boot_handoff_tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -UNDEBUG)
  add_test(NAME boot_update_handoff COMMAND boot_handoff_tests)
  set_tests_properties(boot_update_handoff PROPERTIES TIMEOUT 30)
  add_test(NAME boot_staging_handoff COMMAND boot_handoff_tests --staging)
  set_tests_properties(boot_staging_handoff PROPERTIES TIMEOUT 30)
  add_test(NAME boot_wire_handoff COMMAND boot_handoff_tests --wire-staging)
  set_tests_properties(boot_wire_handoff PROPERTIES TIMEOUT 30)
  add_test(NAME boot_fragment_handoff COMMAND boot_handoff_tests --fragment-staging)
  set_tests_properties(boot_fragment_handoff PROPERTIES TIMEOUT 30)
  add_test(NAME boot_fragment_timing COMMAND boot_handoff_tests --fragment-timing)
  set_tests_properties(boot_fragment_timing PROPERTIES TIMEOUT 30)
  add_test(NAME boot_egress_handoff COMMAND boot_handoff_tests --egress-staging)
  set_tests_properties(boot_egress_handoff PROPERTIES TIMEOUT 30)
  add_test(NAME boot_driver_events_handoff COMMAND boot_handoff_tests --driver-events-staging)
  set_tests_properties(boot_driver_events_handoff PROPERTIES TIMEOUT 30)
  add_test(NAME boot_channel_handoff COMMAND boot_handoff_tests --channel-staging)
  set_tests_properties(boot_channel_handoff PROPERTIES TIMEOUT 30)
  add_test(NAME boot_metadata_setup_handoff COMMAND boot_handoff_tests --metadata-setup-staging)
  set_tests_properties(boot_metadata_setup_handoff PROPERTIES TIMEOUT 30)
  add_test(NAME boot_metadata_handoff COMMAND boot_handoff_tests --metadata-staging)
  set_tests_properties(boot_metadata_handoff PROPERTIES TIMEOUT 30)
  add_test(NAME boot_release_handoff COMMAND boot_handoff_tests --release)
  set_tests_properties(boot_release_handoff PROPERTIES TIMEOUT 30)
  add_executable(boot_profile_tests tests/image_profile_tests.c)
  target_link_libraries(boot_profile_tests PRIVATE tv4_boot_admission)
  target_compile_options(boot_profile_tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -UNDEBUG)
  add_test(NAME boot_image_profile COMMAND boot_profile_tests)
  set_tests_properties(boot_image_profile PROPERTIES TIMEOUT 15)
  add_executable(boot_release_tests tests/release_tests.c)
  target_include_directories(boot_release_tests PRIVATE core)
  target_link_libraries(boot_release_tests PRIVATE tv4_boot_admission tv4_boot_crypto OpenSSL::Crypto)
  target_compile_options(boot_release_tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -UNDEBUG)
  add_test(NAME boot_release_admission COMMAND boot_release_tests)
  set_tests_properties(boot_release_admission PROPERTIES TIMEOUT 30)
  add_test(NAME boot_release_versions COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/release_versions.py" --driver $<TARGET_FILE:boot_release_tests>)
  set_tests_properties(boot_release_versions PROPERTIES TIMEOUT 30)
endif()
